### 🔌 Step 6: Connect Flask to MySQL

In `app/__init__.py`:
```python
from flask import Flask

def create_app():
    app = Flask(__name__)
    app.config['DB_CONFIG'] = {
        'host': 'localhost',
        'user': 'yourusername',
        'password': 'yourpassword',
        'database': 'schemes_db'
    }
    return app
```

In `app/routes.py`:
```python
from flask import render_template
from app import create_app
import mysql.connector

app = create_app()

@app.route('/')
def index():
    db = mysql.connector.connect(**app.config['DB_CONFIG'])
    cursor = db.cursor(dictionary=True)
    cursor.execute("SELECT * FROM schemes")
    schemes = cursor.fetchall()
    db.close()
    return render_template('index.html', schemes=schemes)
```

### 🧠 Step 2: Modify Your Route Logic (`routes.py`)

Update your route to handle search and filter queries:

```python
from flask import render_template, request
from app import create_app
import mysql.connector

app = create_app()

@app.route('/')
def index():
    query = request.args.get('query', '')
    category = request.args.get('category', '')

    db = mysql.connector.connect(**app.config['DB_CONFIG'])
    cursor = db.cursor(dictionary=True)

    sql = "SELECT * FROM schemes WHERE 1=1"
    params = []

    if query:
        sql += " AND (name LIKE %s OR eligibility LIKE %s OR benefits LIKE %s)"
        keyword = f"%{query}%"
        params.extend([keyword, keyword, keyword])

    if category:
        sql += " AND category = %s"
        params.append(category)

    cursor.execute(sql, params)
    schemes = cursor.fetchall()
    db.close()

    return render_template('index.html', schemes=schemes)
```

### 🧠 Step 2: Add a New Route (`routes.py`)

Create a new route to handle scheme detail pages:

```python
@app.route('/scheme/<int:scheme_id>')
def scheme_detail(scheme_id):
    db = mysql.connector.connect(**app.config['DB_CONFIG'])
    cursor = db.cursor(dictionary=True)
    cursor.execute("SELECT * FROM schemes WHERE id = %s", (scheme_id,))
    scheme = cursor.fetchone()
    db.close()

    if not scheme:
        return "Scheme not found", 404

    return render_template('scheme_detail.html', scheme=scheme)
```
